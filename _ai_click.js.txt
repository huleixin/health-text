document.getElementById('qaSmartRecordHost')?.addEventListener('click',(e)=>{
    if(e.target.closest('.ai-record-split-handle')) return;
    const target=e.target.closest('[data-ai-record]');
    if(!target||!document.getElementById('qaSmartRecordHost')?.contains(target)) return;
    const kind=target.getAttribute('data-ai-record');
    const main=target.closest('.ai-record-main.is-split');
    if(main){
      if(main.dataset.aiExpanded!==kind){
        expandAIRecordSide(main,kind);
        return;
      }
    }
    if(kind==='order') openMappedTarget('photo-order');
    else if(kind==='food') openMappedTarget('photo');
  });
  document.getElementById('qaSmartRecordHost')?.addEventListener('keydown',(e)=>{
    if(e.key!=='Enter'&&e.key!==' ') return;
    if(e.target.closest('.ai-record-split-handle')) return;
    const target=e.target.closest('[data-ai-record]');
    if(!target||!document.getElementById('qaSmartRecordHost')?.contains(target)) return;
    e.preventDefault();
    const kind=target.getAttribute('data-ai-record');
    const main=target.closest('.ai-record-main.is-split');
    if(main){
      if(main.dataset.aiExpanded!==kind){
        expandAIRecordSide(main,kind);
        return;
      }
    }
    if(kind==='order') openMappedTarget('photo-order');
    else if(kind==='food') openMappedTarget('photo');
  });
