// --- AI Smart Record V2: fixed icon anchors + absolute split ---
let _aiRecordExpanded='food'; /* 'food' | 'order' */
let _aiSplitXRatio=0.72; /* 0..1 within usable track */

function getAIRecordMetrics(stage){
  const w=stage.clientWidth||1;
  const cs=getComputedStyle(stage);
  const pad=parseFloat(cs.getPropertyValue('--ai-pad'))||8;
  const icon=parseFloat(cs.getPropertyValue('--ai-icon'))||36;
  const gap=parseFloat(cs.getPropertyValue('--ai-gap'))||10;
  const minX=pad+icon+gap; /* just after food icon */
  const maxX=Math.max(minX+24, w-pad-icon-gap); /* just before order icon */
  const foodSnapX=Math.min(maxX, Math.max(minX, w*0.72));
  const orderSnapX=minX;
  return {w,pad,icon,gap,minX,maxX,foodSnapX,orderSnapX};
}
function applyAIRecordSplitX(main,xPx,{dragging=false,ease=false}={}){
  const stage=main?.querySelector('.ai-record-stage');
  if(!stage) return;
  const m=getAIRecordMetrics(stage);
  const x=Math.min(m.maxX, Math.max(m.minX, xPx));
  _aiSplitXRatio=x/m.w;
  /* Blend text by position between snaps */
  const span=Math.max(1, m.foodSnapX-m.orderSnapX);
  const t=Math.min(1, Math.max(0, (x-m.orderSnapX)/span)); /* 0=order snap, 1=food snap */
  const foodText=_aiRecordExpanded==='food'?Math.max(t, dragging?t:1): (dragging?t:0);
  const orderText=_aiRecordExpanded==='order'?Math.max(1-t, dragging?(1-t):1): (dragging?(1-t):0);
  /* While dragging, show blend; when settled, full show for expanded only */
  let ft=foodText, ot=orderText;
  if(!dragging){
    ft=_aiRecordExpanded==='food'?1:0;
    ot=_aiRecordExpanded==='order'?1:0;
  }
  stage.classList.toggle('is-dragging',!!dragging);
  if(ease) stage.classList.add('is-easing');
  stage.style.setProperty('--ai-split-x', x+'px');
  stage.style.setProperty('--ai-food-text', ft.toFixed(3));
  stage.style.setProperty('--ai-order-text', ot.toFixed(3));
  main.dataset.aiExpanded=_aiRecordExpanded;
  if(ease) window.setTimeout(()=>stage.classList.remove('is-easing'),200);
}
function snapAIRecordSplit(main,{ease=true}={}){
  const stage=main.querySelector('.ai-record-stage');
  if(!stage) return;
  const m=getAIRecordMetrics(stage);
  const cur=(_aiSplitXRatio||0.72)*m.w;
  const mid=(m.foodSnapX+m.orderSnapX)/2;
  if(cur>=mid){
    _aiRecordExpanded='food';
    applyAIRecordSplitX(main, m.foodSnapX, {ease});
  }else{
    _aiRecordExpanded='order';
    applyAIRecordSplitX(main, m.orderSnapX, {ease});
  }
}
function expandAIRecordSide(main,kind){
  if(!main||(kind!=='food'&&kind!=='order')) return;
  _aiRecordExpanded=kind;
  const stage=main.querySelector('.ai-record-stage');
  if(!stage) return;
  const m=getAIRecordMetrics(stage);
  applyAIRecordSplitX(main, kind==='food'?m.foodSnapX:m.orderSnapX, {ease:true});
}
function bindAIRecordSplitInteractions(host){
  const main=host?.querySelector('.ai-record-main.is-split');
  const stage=main?.querySelector('.ai-record-stage');
  const handle=main?.querySelector('.ai-record-split-handle');
  if(!main||!stage||!handle||handle.dataset.splitBound==='1') return;
  handle.dataset.splitBound='1';
  _aiRecordExpanded=main.dataset.aiExpanded==='order'?'order':'food';
  const m0=getAIRecordMetrics(stage);
  applyAIRecordSplitX(main, _aiRecordExpanded==='food'?m0.foodSnapX:m0.orderSnapX);

  let dragging=false, startX=0, startSplit=0, activeId=null;

  const onMove=(clientX)=>{
    if(!dragging) return;
    const m=getAIRecordMetrics(stage);
    applyAIRecordSplitX(main, startSplit+(clientX-startX), {dragging:true});
  };
  const onEnd=(e)=>{
    if(!dragging) return;
    if(activeId!=null&&e&&e.pointerId!=null&&e.pointerId!==activeId&&activeId!=='touch') return;
    dragging=false; activeId=null;
    handle.classList.remove('is-dragging');
    try{if(e&&e.pointerId!=null) handle.releasePointerCapture(e.pointerId)}catch(_){}
    snapAIRecordSplit(main,{ease:true});
  };

  handle.addEventListener('pointerdown',(e)=>{
    if(e.button!=null&&e.button!==0) return;
    e.preventDefault(); e.stopPropagation();
    dragging=true; activeId=e.pointerId; startX=e.clientX;
    startSplit=getAIRecordMetrics(stage).w*_aiSplitXRatio;
    handle.classList.add('is-dragging');
    stage.classList.add('is-dragging');
    try{handle.setPointerCapture(e.pointerId)}catch(_){}
  });
  handle.addEventListener('pointermove',(e)=>{
    if(!dragging||e.pointerId!==activeId) return;
    e.preventDefault(); onMove(e.clientX);
  },{passive:false});
  handle.addEventListener('pointerup',onEnd);
  handle.addEventListener('pointercancel',onEnd);
  handle.addEventListener('touchstart',(e)=>{
    if(dragging||!e.touches?.[0]) return;
    e.preventDefault(); e.stopPropagation();
    dragging=true; activeId='touch'; startX=e.touches[0].clientX;
    startSplit=getAIRecordMetrics(stage).w*_aiSplitXRatio;
    handle.classList.add('is-dragging'); stage.classList.add('is-dragging');
  },{passive:false});
  handle.addEventListener('touchmove',(e)=>{
    if(!dragging||activeId!=='touch'||!e.touches?.[0]) return;
    e.preventDefault(); onMove(e.touches[0].clientX);
  },{passive:false});
  handle.addEventListener('touchend',(e)=>{if(activeId==='touch') onEnd(e)},{passive:false});
  handle.addEventListener('touchcancel',(e)=>{if(activeId==='touch') onEnd(e)},{passive:false});
  handle.addEventListener('click',(e)=>{e.preventDefault();e.stopPropagation()});

  /* Keep layout correct on resize */
  if(typeof ResizeObserver!=='undefined'){
    const ro=new ResizeObserver(()=>{
      const m=getAIRecordMetrics(stage);
      applyAIRecordSplitX(main, _aiRecordExpanded==='food'?m.foodSnapX:m.orderSnapX);
    });
    ro.observe(stage);
  }
}
